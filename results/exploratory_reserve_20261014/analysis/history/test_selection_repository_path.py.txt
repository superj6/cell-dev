"""Focused policy tests use fabricated fitness only, with no genomes or life calls."""

import ast
import copy
import importlib.util
import sys
import time
import types
from dataclasses import dataclass, replace
from pathlib import Path

import numpy as np
import pytest
import torch

HERE = Path(__file__).resolve().parent
ROOT = next(p for p in HERE.parents if (p / "src/evolve.py").is_file())
PRODUCER = ROOT / "outputs/autonomous_exploratory_reserve_20261014"


def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    item = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(item)
    return item


S = module("AH_independent_policy_tests", HERE / "selection.py")
P = module("AH_actual_policy_adapter_tests", PRODUCER / "reserve_adapter.py")


@dataclass
class Eval:
    fitness: torch.Tensor
    trial_scores: torch.Tensor


@dataclass
class Draw:
    seeds: tuple
    worlds: tuple


def native():
    result = types.ModuleType("fabricated_native_selection")
    result.__dict__.update(torch=torch, np=np, time=time, Eval=Eval, replace=replace)
    tree = ast.parse((ROOT / "src/evolve.py").read_text())
    names = {"rank_desc", "recheck_draw", "rows", "pooled", "recheck"}
    chosen = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert len(chosen) == len(names)
    exec(compile(ast.Module(body=chosen, type_ignores=[]), "pinned_native_selection_AST", "exec"), result.__dict__)

    class Lineage:
        def write(self, row):
            self.saved = copy.deepcopy(row)

    result.Lineage = Lineage
    return result


def fixture(mode="reserve", generation=0, ties=False, high_protected=False, parents=None, generator=None):
    seed = 701  # Synthetic tie/lottery key only; never a registered life or genetic seed.
    n = native()
    protected = P.reserve_indices(seed, generation)
    values = np.zeros(256) if ties else -np.arange(256, dtype=np.float64)
    if high_protected:
        values[protected] = 1000
    initial = torch.from_numpy(np.repeat(values[:, None], 3, axis=1))
    extra = torch.from_numpy(np.repeat(values[:, None], 4, axis=1))
    res = Eval(initial.mean(1), initial)
    candidates = torch.arange(256).reshape(256, 1)

    class Evaluator:
        def draw(self, purpose, g, lives):
            return Draw(tuple(range(lives)), (None,) * lives)

        def __call__(self, rows, draw):
            trials = extra[rows[:, 0]]
            return Eval(trials.mean(1), trials)

    ev = Evaluator()
    rng = generator if generator is not None else torch.Generator().manual_seed(seed + S.TIE_SEED)
    baseline_rng = torch.Generator().set_state(rng.get_state())
    baseline = native()
    baseline.rank_desc(res.fitness, baseline_rng)
    ordinary = baseline.recheck(ev, candidates, res, generation, 32, 64, 4, "all", baseline_rng)
    parents = list(range(32)) if parents is None else parents
    child_ids = list(range(32 + 224 * generation, 32 + 224 * (generation + 1)))
    ids = parents + child_ids
    with P.installed(n, seed, mode, parent_ids=parents):
        n.rank_desc(res.fitness, rng)
        rc = n.recheck(ev, candidates, res, generation, 32, 64, 4, "all", rng)
        chosen = rc["order"].tolist()
        row = {
            "gen": generation,
            "children": [
                {
                    "id": value,
                    "pa": parents[0],
                    "pb": parents[1],
                    "ops": {"mut": []},
                    "fitness": float(res.fitness[32 + i]),
                    "kept": 32 + i in chosen,
                }
                for i, value in enumerate(child_ids)
            ],
            "survivors": [ids[i] for i in chosen],
            "fitness": rc["chosen"].fitness.tolist(),
            "recheck": {
                "ids": [ids[i] for i in rc["shortlist"].tolist()],
                "key": "all",
                "F_base": res.fitness[rc["shortlist"]].tolist(),
                "F_recheck": rc["res"].fitness.tolist(),
                "changed": rc["changed"],
            },
        }
        lineage = n.Lineage()
        lineage.write(row)
        row = lineage.saved
    curve = {"recheck_changed": rc["changed"], "best_id": row["survivors"][0], "fitness": row["fitness"][0]}
    if mode == "control" or generation == 3:
        for key in ("order", "shortlist", "pick"):
            assert torch.equal(rc[key], ordinary[key])
        assert torch.equal(rng.get_state(), baseline_rng.get_state())
        assert torch.equal(rc["chosen"].trial_scores, ordinary["chosen"].trial_scores)
    return row, curve, parents, seed, mode


def audit(args):
    row, curve, parents, seed, mode = args
    return S.audit_generation(
        row, curve, parents, seed, mode, P.digest(P.POLICY), torch.Generator().manual_seed(seed + S.TIE_SEED)
    )


@pytest.mark.parametrize(
    "mode,generation,ties,high",
    [
        ("control", 0, False, False),
        ("control", 2, True, False),
        ("reserve", 0, False, False),
        ("reserve", 1, True, False),
        ("reserve", 2, False, True),
        ("reserve", 3, False, False),
        ("reserve", 3, True, False),
    ],
)
def test_actual_native_adapter_selection(mode, generation, ties, high):
    args = fixture(mode, generation, ties, high)
    survivors, receipt = audit(args)
    assert len(set(survivors)) == 32 and receipt["ranking_calls_exact"] == 3
    assert receipt["active"] == (mode == "reserve" and generation < 3)
    assert not {"src.run", "src.life.build", "src.life.evaluate", "src.evolve"} & set(sys.modules)


@pytest.mark.parametrize(
    "corruption", ["lottery", "policy", "candidate_ids", "base", "pooled", "rank", "tie", "short", "pick", "survivor"]
)
def test_policy_corruption_fails(corruption):
    args = fixture()
    row = args[0]
    item = row["exploratory_reserve"]
    if corruption == "lottery":
        item["reserve_indices"][0] = 0
    elif corruption == "policy":
        item["policy_sha256"] = "0" * 64
    elif corruption == "candidate_ids":
        item["candidate_ids"][0] = 99999
    elif corruption == "base":
        item["base_F"][0] += 1
    elif corruption == "pooled":
        item["ranking_calls"][2]["fitness"][0] += 1
    elif corruption == "rank":
        item["ranking_calls"][0]["order"][:2] = reversed(item["ranking_calls"][0]["order"][:2])
    elif corruption == "tie":
        item["ranking_calls"][1]["tie_after_sha256"] = "0" * 64
    elif corruption == "short":
        item["shortlist_indices"][-1] = 0
    elif corruption == "pick":
        item["pick_positions"][-1] = item["pick_positions"][0]
    else:
        row["fitness"][0] += 1
    with pytest.raises(ValueError):
        audit(args)


def test_lottery_rejection_and_distinctness(monkeypatch):
    class Digest:
        def __init__(self, value):
            self.value = value

        def digest(self):
            return self.value.to_bytes(8, "little") + bytes(24)

    values = iter([(1 << 64) - 1, 0, 0])
    monkeypatch.setattr(S.hashlib, "sha256", lambda payload: Digest(next(values)))
    assert S.reserve_positions(1, 0) == [32, 33]


def test_separate_lottery_agrees_with_declared_adapter():
    for seed in (0, 1, 701, (1 << 64) - 1):
        for generation in range(4):
            a = S.reserve_positions(seed, generation)
            assert a == P.reserve_indices(seed, generation)
            assert len(set(a)) == 2 and all(32 <= x < 256 for x in a)


@pytest.mark.parametrize("mode,corrupt", [("control", False), ("reserve", False), ("control", True), ("reserve", True)])
def test_full_four_generation_policy_and_checkpoint_tie_state(mode, corrupt):
    # Repeated synthetic fitness batches exercise accumulated tie state and changing
    # breeder identifiers; no genotype arrays or scientific evaluations exist.
    seed = 701
    rng = torch.Generator().manual_seed(seed + S.TIE_SEED)
    parents = list(range(32))
    lines = [
        {
            "event": "founders",
            "sampled": 0,
            "ids": [0],
            "copies": [{"id": i, "parent": 0, "ops": {"mut": []}} for i in range(1, 32)],
        }
    ]
    curves = []
    spec = types.SimpleNamespace(hash64=lambda seed, purpose, index, life: purpose * 100000 + index * 100 + life)
    for generation in range(4):
        row, curve, _, _, _ = fixture(mode, generation, ties=True, parents=parents, generator=rng)
        row["seeds"] = [S.life_seed(spec, seed, 1, generation, j) for j in range(3)]
        row["recheck"]["seeds"] = [S.life_seed(spec, seed, 1, generation, j) for j in range(3, 7)]
        row["T"] = curve["T"] = 1024
        curve["gen"] = generation
        lines.append(row)
        curves.append(curve)
        parents = row["survivors"]
    state = {
        "ids": torch.tensor(parents),
        "fitness": torch.tensor(lines[-1]["fitness"], dtype=torch.float64),
        "tie_generator": rng.get_state(),
    }
    if corrupt:
        state["tie_generator"] = torch.Generator().manual_seed(1).get_state()
        with pytest.raises(ValueError):
            S.audit_lineage(lines, curves, state, seed, spec, mode, P.POLICY)
    else:
        receipt = S.audit_lineage(lines, curves, state, seed, spec, mode, P.POLICY)
        assert receipt["tie_ranking_calls_replayed"] == 12
        assert [v["active"] for v in receipt["policy_decisions"]] == (
            [True, True, True, False] if mode == "reserve" else [False] * 4
        )
